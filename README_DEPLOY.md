## 2) Supabase: Band members table + RLS

Run this in Supabase SQL Editor for your project (safe/idempotent). It ensures the `band_members` table exists, adds `category`, enables RLS, and sets policies.

```sql
-- Table
create table if not exists public.band_members (
  id bigint primary key generated by default as identity,
  name text not null,
  role text not null,
  image text not null,
  description text not null,
  specialty text not null,
  achievements text[] default '{}',
  order_position integer default 1,
  enabled boolean default true,
  is_featured boolean default false,
  social_links jsonb default '{}',
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- Ensure category column exists
do $$
begin
  if not exists (
    select 1 from information_schema.columns
    where table_schema = 'public' and table_name = 'band_members' and column_name = 'category'
  ) then
    alter table public.band_members add column category text default 'Vocals' not null;
  end if;
end $$;

alter table public.band_members enable row level security;

-- Policies (drop/create to ensure correctness)
drop policy if exists "Enable read access for all users" on public.band_members;
drop policy if exists "Enable insert for authenticated users" on public.band_members;
drop policy if exists "Enable update for authenticated users" on public.band_members;
drop policy if exists "Enable delete for authenticated users" on public.band_members;

create policy "Enable read access for all users"
  on public.band_members for select to public using (true);

create policy "Enable insert for authenticated users"
  on public.band_members for insert to authenticated with check (true);

create policy "Enable update for authenticated users"
  on public.band_members for update to authenticated using (true) with check (true);

create policy "Enable delete for authenticated users"
  on public.band_members for delete to authenticated using (true);
```

## 3) Supabase Storage: `band-images` bucket + policies

Uploads for band member photos use bucket `band-images`. Run this to create/ensure it with correct policies.

```sql
-- Bucket
insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)

on conflict (id) do update
set file_size_limit = excluded.file_size_limit,
    allowed_mime_types = excluded.allowed_mime_types,
    public = excluded.public;

-- Policies
drop policy if exists "Public read band-images" on storage.objects;
drop policy if exists "Auth insert band-images" on storage.objects;
drop policy if exists "Auth update band-images" on storage.objects;
drop policy if exists "Auth delete band-images" on storage.objects;

create policy "Public read band-images" on storage.objects for select to public using (bucket_id = 'band-images');
create policy "Auth insert band-images" on storage.objects for insert to authenticated with check (bucket_id = 'band-images');
create policy "Auth update band-images" on storage.objects for update to authenticated using (bucket_id = 'band-images') with check (bucket_id = 'band-images');
create policy "Auth delete band-images" on storage.objects for delete to authenticated using (bucket_id = 'band-images');
```
